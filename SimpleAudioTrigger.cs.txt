using UnityEngine;

namespace ShowroomVR
{
    /// <summary>
    /// Simple audio trigger for player.
    /// Add this to an object with a BoxCollider (Is Trigger = true).
    /// </summary>
    [RequireComponent(typeof(AudioSource))]
    public class SimpleAudioTrigger : MonoBehaviour
    {
        [Tooltip("Play sound only once?")]
        public bool playOnce = false;

        [Tooltip("Tag of the player object to detect")]
        public string playerTag = "Player";

        private AudioSource audioSource;
        private bool hasPlayed = false;

        void Start()
        {
            audioSource = GetComponent<AudioSource>();
        }

        void OnTriggerEnter(Collider other)
        {
            if (!other || !gameObject.activeInHierarchy) return;

            if (playOnce && hasPlayed) return;

            if (string.IsNullOrEmpty(playerTag) || other.CompareTag(playerTag))
            {
                if (audioSource != null && !audioSource.isPlaying)
                {
                    audioSource.Play();
                    hasPlayed = true;
                }
            }
        }

        void OnDisable()
        {
            hasPlayed = false;
        }
    }
}
